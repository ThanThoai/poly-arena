import requests

url     = ""
headers = {
    "Content-Type": "application/json",
    "x-api-key":     "YOUR_API_KEY",
}
payload = {
    "symbol":    "BTC",
    "timeframe": "M5",
    "forecast":  "GREEN",
    "amount":    100,
}

res  = requests.post(url, headers=headers, json=payload)
print(res.json())
